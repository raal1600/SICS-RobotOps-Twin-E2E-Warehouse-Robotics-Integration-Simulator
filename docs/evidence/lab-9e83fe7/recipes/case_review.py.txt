"""Prepared offline saved-case reviewer; pinned source/campaign, explicit pass and fixture required."""
import argparse
import sys
import collections,hashlib,json,re,subprocess
from datetime import datetime,timezone
from pathlib import Path
from urllib.parse import urlsplit
import xml.etree.ElementTree as ET

def review(pass_number, fixture):
    r=Path(__file__).resolve().parents[3]; out=r/'artifacts/final-evidence-pack/9e83fe7'; campaign=r/'docs/evidence/acceptance/20261006T024719'
    expected={'commit':'9e83fe7d41a52a63ba65b77730e392dafca8fffe','dirty':False,'source_sha256':'136641a25c5fbd7d45c016582ec2c5c5197362f44f0a894ee95a69c652e3eafd'}
    if pass_number not in (1,2) or not re.fullmatch(r'robotops-acceptance-'+str(pass_number)+r'-[A-Za-z0-9_]+',fixture):
     raise ValueError('Expected exact observed fixture basename for this pass')
    names=['happy','drop_ack_after_effect','duplicate_delivery','wms_unavailable']
    faults={n:(None if n=='happy' else n.upper()) for n in names}
    issues=[]; raw_inputs={}; secret_findings=[]; checks_total=0
    def h(p): return hashlib.sha256(p.read_bytes()).hexdigest()
    def byte_ref(p,raw): return {'path':p.relative_to(r).as_posix(),'sha256':hashlib.sha256(raw).hexdigest(),'bytes':len(raw)}
    def read_bound(p):
     raw=p.read_bytes(); item=byte_ref(p,raw); previous=raw_inputs.get(item['path'])
     if previous is not None and previous!=item: raise ValueError('Bound input changed: '+item['path'])
     raw_inputs[item['path']]=item
     return raw
    def ref(p):
     read_bound(p)
     return raw_inputs[p.relative_to(r).as_posix()]
    def load(p):
     raw=read_bound(p)
     return json.loads(raw.decode('utf-16' if raw.startswith((b'\xff\xfe',b'\xfe\xff')) else 'utf-8-sig'))
    def walk_secret(v,path,p):
     if isinstance(v,dict):
      for k,x in v.items():
       if re.fullmatch(r'(?i)(password|passwd|api[_-]?key|access[_-]?token|client[_-]?secret|authorization)',k) and isinstance(x,str) and x and not re.fullmatch(r'(?i)(redacted|\*+|<redacted>|\[redacted\])',x): secret_findings.append({'path':path,'field':p+'.'+k,'kind':'sensitive_key_value'})
       walk_secret(x,path,p+'.'+k)
     elif isinstance(v,list):
      for i,x in enumerate(v): walk_secret(x,path,p+f'[{i}]')
     elif isinstance(v,str):
      for kind,pattern in [('fixture_credential',r'demo-only|local-demo-only|guest:guest|postgresql://'),('url_userinfo',r'\b[a-z][a-z0-9+.-]*://[^/\s:]+:[^@\s/]+@'),('bearer_or_private_key',r'(?i)\bBearer\s+[A-Za-z0-9._~+/-]{12,}|-----BEGIN (?:RSA |EC |OPENSSH )?PRIVATE KEY-----')]:
       if re.search(pattern,v): secret_findings.append({'path':path,'field':p,'kind':kind})
    def scan(v,p): walk_secret(v,p.relative_to(r).as_posix(),'$')
    # Refuse before any writes unless the exact pass is terminal and all four saved cases are formally passed.
    def required(condition,label):
     if not condition: raise ValueError(label)
    def plain_json(raw): return json.loads(raw.decode('utf-16' if raw.startswith((b'\xff\xfe',b'\xfe\xff')) else 'utf-8-sig'))
    def pinned_source():
     def git(*args): return subprocess.check_output(['git',*args],cwd=r,text=True,encoding='utf-8').strip()
     checksum=hashlib.sha256()
     for name in sorted(set(git('ls-files','--cached','--others','--exclude-standard').splitlines())):
      path=r/name
      if path.is_file() and not name.startswith('docs/evidence/') and name!='ACCEPTANCE_REPORT.md': checksum.update(name.encode()+b'\0'+path.read_bytes())
     required(git('rev-parse','HEAD')==expected['commit'] and checksum.hexdigest()==expected['source_sha256'],'current source commit/fingerprint changed')
     return {'commit':expected['commit'],'source_sha256':checksum.hexdigest(),'dirty':bool(git('status','--porcelain'))}
    current_source=pinned_source()
    snapshot_dir=out/f'pass{pass_number}-finalized-snapshots'; target=out/f'pass{pass_number}-case-review-final.json'
    required(not snapshot_dir.exists() and not target.exists(),'Existing immutable snapshots or review preserved; inspect before a new run')
    live_manifest=r/'docs/evidence/acceptance/20261006T024719/manifest.json'
    live_status=out/'status.json'
    mutable_paths=[out/f'pass-{pass_number}'/f'{name}.json' for name in names]+[live_status,live_manifest]
    required(all(p.is_file() for p in mutable_paths),'Four case snapshots/status/manifest not all available')
    frozen_bytes={p:p.read_bytes() for p in mutable_paths}
    frozen_refs={p:byte_ref(p,raw) for p,raw in frozen_bytes.items()}
    pre_manifest=plain_json(frozen_bytes[live_manifest]); pre_status=plain_json(frozen_bytes[live_status])
    required(pre_manifest['source']==expected and pre_manifest['started_at']=='2026-10-06T02:47:20.134216+00:00','Wrong campaign source/start')
    required(pre_status['source']==expected and pre_status['acceptance_directory']==campaign.relative_to(r).as_posix() and pre_status['fixture_roots'][str(pass_number)]==fixture,'Watcher source/campaign/selected pass fixture differs')
    gate=pre_manifest['gates'].get(f'tests_{pass_number}',{})
    required(gate.get('passed') is True and type(gate.get('exit_code')) is int and gate['exit_code']==0,'Selected full-suite gate has not exited cleanly')
    command=gate['command']
    basetemp=[arg.split('=',1)[1] for arg in command if arg.startswith('--basetemp=')]
    junitflags=[arg.split('=',1)[1] for arg in command if arg.startswith('--junitxml=')]
    required(len(basetemp)==len(junitflags)==1 and Path(basetemp[0]).name=='fixtures' and Path(basetemp[0]).parent.name==fixture and Path(junitflags[0]).resolve()==campaign/f'tests-{pass_number}.xml','Selected pass command/fixture/JUnit binding differs')
    required(command[1:4]==['-m','tools.dev','test'] and len(command)==10 and command[-1]=='--cov-report=term' and command[6:8]==['--cov=robotops','--cov=apps'],'Unexpected filtered/full-suite command')
    terminal_ids=[]
    for index in range(1,pass_number+1):
     jp=campaign/f'tests-{index}.xml'
     required(jp.is_file(),'Required terminal JUnit missing')
     raw=read_bound(jp); doc=ET.fromstring(raw); suites=list(doc.iter('testsuite')); cases=list(doc.iter('testcase'))
     required(bool(suites) and bool(cases),'Empty JUnit')
     required(not any(list(s.iter('testsuite'))[1:] for s in suites),'Unsupported nested JUnit suites')
     required(all(int(s.attrib['tests'])==len(s.findall('testcase')) and all(int(s.attrib[k])==0 for k in ['failures','errors','skipped']) for s in suites),'Declared JUnit counts/errors/skips differ')
     required(not any(list(doc.iter(k)) for k in ['failure','error','skipped']),'JUnit contains failed/error/skipped cases')
     ids=[c.attrib['classname'].replace('.','/')+'.py::'+c.attrib['name'] for c in cases]
     required(len(ids)==len(set(ids)) and sum(int(s.attrib['tests']) for s in suites)==len(ids),'Duplicate/unowned JUnit cases')
     required(all('tests/lab/test_browser_lab.py::test_full_lab_browser_authorization_recovery_and_business_completion['+(faults[name] or '')+']' in ids for name in names),'Exact lab cases missing')
     terminal_ids.append(set(ids))
    if pass_number==2: required(terminal_ids[0]==terminal_ids[1],'Full pass1/pass2 JUnit testcase sets differ')
    coverage_path=campaign/f'coverage-{pass_number}.json'
    required(coverage_path.is_file(),'Selected terminal-suite coverage missing')
    coverage_totals=load(coverage_path)['totals']
    required(coverage_totals['num_statements']>0 and coverage_totals['covered_lines']/coverage_totals['num_statements']*100>=85 and abs(coverage_totals['covered_lines']/coverage_totals['num_statements']*100-coverage_totals['percent_covered'])<1e-8,'Coverage invalid/below85')
    for name in names:
     saved=plain_json(frozen_bytes[out/f'pass-{pass_number}'/f'{name}.json'])
     required(saved['formal_test_status']=='PASSED' and saved['formal_test_result']['junit']==(campaign/f'tests-{pass_number}.xml').relative_to(r).as_posix() and saved['formal_test_result']['name']=='test_full_lab_browser_authorization_recovery_and_business_completion['+(faults[name] or '')+']','Watcher formal result not promoted/bound')
     required(saved['capture_complete'] is True and saved['acceptance_pass']==pass_number and saved['source']==expected and saved['acceptance_directory']==campaign.relative_to(r).as_posix(),'Case source/pass/capture binding differs')
     required(Path(saved['stack']['data_directory']).is_relative_to(Path(basetemp[0])) and fixture in Path(saved['stack']['data_directory']).parts,'Saved case outside exact selected fixture')
     artifact_path=(r/saved['artifact_directory']).resolve()
     required(artifact_path.is_relative_to(r/'artifacts/lab-browser'),'Raw artifacts outside lab-browser scope')
     expected_inputs={artifact_path/name for name in ['stack.json','session.json','browser.json','source-hashes.json']}
     required(len(saved['input_files'])==4 and {(r/item['path']).resolve() for item in saved['input_files']}==expected_inputs,'Unexpected raw input scope; no database/ZIP reads allowed')
    for p,raw in frozen_bytes.items(): required(p.read_bytes()==raw,'Mutable input changed before immutable copy')
    snapshot_dir.mkdir()
    copy_entries=[]
    for p,raw in frozen_bytes.items():
     destination=snapshot_dir/p.name
     with destination.open('xb') as stream: stream.write(raw)
     raw_inputs[destination.relative_to(r).as_posix()]=byte_ref(destination,raw)
     required(read_bound(destination)==raw,'Copy byte verification failed')
     copy_entries.append({'source':frozen_refs[p]['path'],'copy':destination.relative_to(r).as_posix(),'sha256':frozen_refs[p]['sha256'],'bytes':frozen_refs[p]['bytes']})
    copy_receipt={'copied_at_utc':datetime.now(timezone.utc).isoformat(),'source':expected,'acceptance_pass':pass_number,'fixture_basename':fixture,'files':copy_entries,'purpose':'Fixed formal-PASSED input binding; watcher cannot mutate these copies. No product/test/runtime actions.'}
    with (snapshot_dir/'copy-receipt.json').open('x',encoding='utf-8',newline='\n') as stream: json.dump(copy_receipt,stream,indent=2);stream.write('\n')
    load(snapshot_dir/'copy-receipt.json')
    manifest=load(snapshot_dir/'manifest.json'); watcher=load(snapshot_dir/'status.json')
    assert manifest['source']==expected and watcher['source']==expected
    assert watcher['acceptance_directory']==campaign.relative_to(r).as_posix() and watcher['fixture_roots'][str(pass_number)]==fixture
    assert subprocess.check_output(['git','rev-parse','HEAD'],cwd=r,text=True).strip()==expected['commit']
    junit_path=campaign/f'tests-{pass_number}.xml'; formal={}; suite={'status':'PENDING_JUNIT','path':junit_path.relative_to(r).as_posix(),'file_exists_at_review':junit_path.exists()}
    if junit_path.exists():
     try:
      xml=ET.fromstring(read_bound(junit_path)); allcases=list(xml.iter('testcase')); declared=list(xml.iter('testsuite'))
      if allcases and declared:
       keys=[c.attrib['classname'].replace('.','/')+'.py::'+c.attrib['name'] for c in allcases]
       outcomes=[next((s for s in ['error','failure','skipped'] if c.find(s) is not None),'passed') for c in allcases]
       counts=dict(collections.Counter(outcomes)); duplicates=[k for k,n in collections.Counter(keys).items() if n>1]
       formal=dict(zip(keys,outcomes))
       suite={'status':'TERMINAL_JUNIT','artifact':ref(junit_path),'actual_testcase_count':len(allcases),'unique_testcase_count':len(set(keys)),'outcomes':counts,'duplicates':duplicates,'declared_suites':[d.attrib for d in declared],'manifest_gate':manifest['gates'].get(f'tests_{pass_number}')}
       suite['declared_counts_match']=sum(int(d.attrib['tests']) for d in declared)==len(allcases)
       suite['clean']=not duplicates and suite['declared_counts_match'] and counts.get('passed')==len(allcases) and all(int(d.attrib[k])==0 for d in declared for k in ['errors','failures','skipped'])
       if not suite['clean']: issues.append({'scope':f'suite_{pass_number}','issue':'Terminal JUnit is not wholly clean; exact counts retained.'})
       cp=campaign/f'coverage-{pass_number}.json'
       if cp.exists(): cov=load(cp); suite['coverage']={'artifact':ref(cp),'totals':cov['totals']}
     except ET.ParseError:
      suite['status']='PENDING_COMPLETE_JUNIT_XML'
    results={}; source_sets=[]
    for n in names:
     p=snapshot_dir/f'{n}.json'; d=load(p); scan(d,p)
     a=r/d['artifact_directory']; s=load(a/'session.json'); b=load(a/'browser.json'); st=load(a/'stack.json'); sh=load(a/'source-hashes.json')
     for fn,obj in [('session.json',s),('browser.json',b),('stack.json',st),('source-hashes.json',sh)]: scan(obj,a/fn)
     checks={}
     def check(k,v):
      nonlocal checks_total
      checks_total+=1;checks[k]=bool(v)
      if not v: issues.append({'case':n,'check':k})
     for item in d['input_files']: check('input_hash_'+Path(item['path']).name,h(r/item['path'])==item['sha256'])
     source_set={(r/Path(k)).resolve().relative_to(r).as_posix():v for k,v in sh.items()}; source_sets.append(source_set)
     snapshot_set={x['path'].replace('\\','/'):x['recorded_sha256'] for x in d['source_checks']}
     check('source_manifest_exact',d['source']==expected)
     check('campaign_pass_fixture_exact',d['acceptance_pass']==pass_number and d['acceptance_directory']==campaign.relative_to(r).as_posix() and fixture in Path(st['data_directory']).parts)
     check('capture_complete',d['capture_complete'] is True)
     check('raw_stack_matches',st==d['stack'])
     check('source_map_matches_snapshot',source_set==snapshot_set and len(source_set)==35)
     check('raw_session_matches_summary',all(s[k]==v for k,v in d['session'].items()))
     sid,cid,jid=s['session_id'],s['command_id'],s['job_id']
     fault=faults[n]; runtime='blender' if n=='happy' else 'synthetic'
     check('runtime_truthful',d['runtime']==st['runtime']==runtime)
     check('lab_guided_original_fault',s['profile']=='lab' and s['mode']=='guided' and s['fault']==fault)
     check('completed_stage22',s['status']=='COMPLETED' and s['current_stage']==22 and s['pending_authorization'] is None)
     check('revision_steps',s['revision']==(46 if n in ['drop_ack_after_effect','wms_unavailable'] else 44) and len(s['steps'])==(23 if n in ['drop_ack_after_effect','wms_unavailable'] else 22))
     steps=s['steps']; bystage={k:[x for x in steps if x['stage']==k] for k in range(1,23)}
     check('all22_stages_present',all(bystage[k] for k in bystage))
     check('all_steps_bound_original_session',all(x['session_id']==sid and x['correlation_id']==s['correlation_id'] and x['command_id'] in [None,cid] for x in steps))
     runtime_db=d['journals']['runtime.db']; plc=d['journals']['plc.db']
     check('journal_paths_exact_fixture',all(fixture in Path(x['path']).parts and Path(x['path']).parent==Path(st['data_directory']) and x['read_only'] is True for x in [runtime_db,plc]))
     check('one_row_each_journal',len(runtime_db['controller_journal'])==len(plc['plc_commands'])==len(plc['plc_results'])==1)
     row=runtime_db['controller_journal'][0]; receipt=row['receipt']; pc=plc['plc_commands'][0]; pr=plc['plc_results'][0]; payload=row['payload_hash']
     check('original_command_payload_all_journals',row['command_id']==pc['command_id']==pr['command_id']==receipt['command_id']==cid and pc['payload_hash']==receipt['payload_hash']==payload)
     check('receipt_job_correlation',receipt['job_id']==jid and receipt['correlation_id']==s['correlation_id'])
     check('exact_original_receipt_all_journals',receipt==pc['result']==pr['receipt'])
     check('one_durable_effect',receipt['effect_count']==d['physical_effect_count']==1 and receipt['journal_durable'] is True and receipt['status']=='SUCCEEDED')
     check('plc_succeeded_sequence1',pc['state']=='SUCCEEDED' and pc['result_sequence']==pr['sequence']==1)
     check('runtime_receipt_reason',receipt['reason']==('BLENDER_PICK_APPLIED' if n=='happy' else 'PICK_APPLIED'))
     requests=b['requests']; posts=[x for x in requests if x['method']=='POST']
     physical=[x for x in posts if (x.get('body') or {}).get('stage')==16]
     consent=[x for x in posts if (x.get('body') or {}).get('stage')==15]
     reconciles=[x for x in posts if urlsplit(x['url']).path.endswith('/reconcile')]
     businessposts=[x for x in posts if (x.get('body') or {}).get('stage')==20]
     creations=[x for x in posts if urlsplit(x['url']).path=='/v1/wms/tasks']
     check('original_creation_request',len(creations)==1 and creations[0]['body']['fault']==fault and creations[0]['body']['request']['order_id']==s['order_id'])
     check('one_browser_physical_post',len(physical)==1 and physical[0]['url'].endswith('/'+sid+'/authorize'))
     check('consent_before_physical',len(consent)==1 and requests.index(consent[0])<requests.index(physical[0]) and bystage[15][0]['output']['authorized'] is True and bystage[15][0]['output']['command_id']==cid and bystage[15][0]['output']['payload_hash']==payload)
     check('no_run_endpoint',not any(urlsplit(x['url']).path.endswith('/run') for x in posts))
     check('browser_no_errors',not b['errors'] and not b['console_errors'] and b['errors']==d['browser_errors'] and b['console_errors']==d['browser_console_errors'])
     pre=s['context']['execution_preflight']
     check('preflight_original_READY_no_result',pre['command_id']==cid and pre['payload_hash']==payload and pre['state']=='READY' and pre['result_sequence']==0 and pre['result'] is None)
     check('truthful_classifications',all(x['classification']=='REAL PROTOCOL' for x in steps if x['stage'] in [10,11,12,13,14,17,20]) and bystage[16][0]['classification']=='SIMULATED ROBOT')
     opc=bystage[12][0]['output']
     check('real_opc_edge_distinct_process',opc['opcua_client_component']=='edge-adapter' and opc['edge_process_id']==st['processes']['edge'] and opc['edge_process_id']!=st['processes']['api'])

     business=bystage[20]; lastbusiness=business[-1]['output']; erp=bystage[21][-1]['output']
     check('wms_final200_no_physical_retry',lastbusiness['wire']['http_status']==200 and lastbusiness['wms_acknowledged'] is True and lastbusiness['physical_retry'] is False and lastbusiness['command_id']==cid and lastbusiness['job_id']==jid)
     check('erp_complete',erp['business_status']=='COMPLETED' and erp['wms_acknowledged'] is True and erp['order_id']==s['order_id'] and s['context']['wms_ack'] is True)

     detail={}
     if n=='happy':
      motion=load(a/'blender-motion.json'); scan(motion,a/'blender-motion.json'); rec=motion['recording']

      check('blender_original_complete',rec['source']=='BLENDER_EVALUATED_SCENE' and rec['complete'] is True and rec['command_id']==cid)
      check('actual_133frames_118poses',rec['frames']==rec['total_frames']==rec['motion_steps']==133 and rec['robot_poses']==118)
      check('product_moved',rec['product_start']!=rec['product_finish'])
      check('pose_and_rendered_movement',motion['after']['playing'] is True and motion['after']['cursor']>motion['before']['cursor'] and motion['after']['pose']!=motion['before']['pose'] and motion['after']['rendered']!=motion['before']['rendered'] and d['motion']['rendered_position_changed'] is True)

     if n=='drop_ack_after_effect':
      recovery=[x for x in steps if x['wire'].get('original_identity_queried') is True]
      check('unknown_receipt_loss',bystage[16][0]['output']['status']=='UNKNOWN_OUTCOME' and bystage[16][0]['output']['receipt'] is None and bystage[19][0]['state_after']=='UNKNOWN_OUTCOME')
      check('one_original_reconcile_no_resend',len(reconciles)==len(recovery)==1 and recovery[0]['state_before']=='UNKNOWN_OUTCOME' and recovery[0]['state_after']=='COMPLETED' and recovery[0]['wire']['command_id']==cid and recovery[0]['wire']['physical_resend'] is False and recovery[0]['output']['command']['command_id']==cid and recovery[0]['output']['receipt']==receipt)

     else: check('no_reconcile_post',len(reconciles)==0)
     delivery=bystage[11][0]['output']
     if n=='duplicate_delivery':

     check('watcher_no_secret_findings',d['secret_review']['potential_findings']==[])

def main(pass_number):
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--review-terminal-pass'+str(pass_number), action='store_true', dest='opt_in')
    parser.add_argument('--expected-fixture', required=True, help='Exact observed basename, cross-checked with watcher and terminal command')
    args=parser.parse_args()
    if not args.opt_in:
        raise ValueError('Preparation only; explicit terminal pass review opt-in required')
    review(pass_number,args.expected_fixture)

def entry(pass_number):
    try:
        main(pass_number)
    except (ValueError,AssertionError,KeyError,TypeError,IndexError,OSError,ET.ParseError,subprocess.CalledProcessError) as exc:

        sys.exit(2)
