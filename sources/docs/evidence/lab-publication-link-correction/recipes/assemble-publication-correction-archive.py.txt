"""Archive saved publication correction evidence without executing verification recipes."""
import hashlib
import json
from datetime import datetime, timezone
from pathlib import Path
import xml.etree.ElementTree as ET

ROOT = Path(__file__).resolve().parents[3]
PROOF = ROOT / "artifacts/final-evidence-pack/9e83fe7/publication-link-correction"
OUT = ROOT / "docs/evidence/lab-publication-link-correction"
SITE = ROOT / "docs/evidence/lab-9e83fe7/publication/site"
A = "9e83fe7d41a52a63ba65b77730e392dafca8fffe"
cache = {}
def raw(p):
    p = p.resolve()
    if p not in cache:
        cache[p] = p.read_bytes()
    return cache[p]
def sha(b):
    return hashlib.sha256(b).hexdigest()
def rel(p):
    return p.resolve().relative_to(ROOT).as_posix()
def ref(p):
    b = raw(p)
    return {"path": rel(p), "absolute_path": str(p.resolve()), "sha256": sha(b), "bytes": len(b)}
def load(p):
    return json.loads(raw(p))
def encode(v):
    return (json.dumps(v, ensure_ascii=False, indent=2) + "\n").encode()
assert ROOT.name == "robotops-twin"
if OUT.exists():
    raise SystemExit("Refusing existing archive.")
bindings = load(PROOF / "source-snapshot-bindings.json")
final = load(PROOF / "final-proposed-report-render-review.json")
focused = load(PROOF / "correction-receipt.json")
saved = load(PROOF / "saved-site-checks.json")
assert final["proposed_report"]["sha256"] == "98b583b8e1fbe2dbf09158df42396980ac407ebe17292b5cf77912be3d2a5df4"
assert final["all7candidate_HTMLs_validated_by_actual_new_validate_links"]
assert final["actual_build_report_AST_used"] and final["only_line2_changed"]
assert focused["source_A"] == A
suites = []
for label, expect in (("baseline", (3, 2, 0, 0)), ("proposed", (5, 0, 0, 0))):
    cases = list(ET.fromstring(raw(PROOF / (label + ".xml"))).iter("testcase"))
    counts = (sum(not any(c.find(t) is not None for t in ("failure", "error", "skipped")) for c in cases),
              sum(c.find("failure") is not None for c in cases),
              sum(c.find("error") is not None for c in cases),
              sum(c.find("skipped") is not None for c in cases))
    assert len(cases) == 5 and counts == expect
    suites.append(dict(label=label, testcases=5, passed=counts[0], failures=counts[1],
                       errors=counts[2], skips=counts[3],
                       exit_code=load(PROOF / (label + "-exit.json"))["exit_code"]))
ruff = load(PROOF / "ruff-result.json")
assert ruff["ruff_check_exit"] == ruff["ruff_format_exit"] == 0
copies = []
def plan(p, dest, kind):
    r = ref(p)
    assert not any(x["archive_path"] == rel(OUT / dest) for x in copies)
    copies.append(dict(source_path=r["path"], source_absolute_path=r["absolute_path"],
                       archive_path=rel(OUT / dest), sha256=r["sha256"], bytes=r["bytes"], kind=kind))
for a in bindings["aliases"]:
    p = ROOT / a["snapshot_path"]
    assert sha(raw(p)) == a["recorded_sha256"] and len(raw(p)) == a["bytes"]
    plan(p, "sources/" + p.name, "exact_inert_source_snapshot")
proof_names = [
    "baseline.xml", "baseline-exit.json", "baseline.stdout.log", "baseline.stderr.log",
    "proposed.xml", "proposed-exit.json", "proposed.stdout.log", "proposed.stderr.log",
    "ruff-result.json", "ruff-check.stdout.log", "ruff-check.stderr.log",
    "ruff-format.stdout.log", "ruff-format.stderr.log", "initial-ruff-observation.json",
    "build-publication.diff.txt", "exact-two-href.diff.txt", "correction-receipt.json",
    "saved-site-checks.json", "final-proposed-report-render-review.json", "source-snapshot-bindings.json"]
for n in proof_names:
    plan(PROOF / n, "proof/" + n, "exact_original_result_or_receipt")
plan(PROOF / "verify_saved_site.py", "recipes/verify_saved_site.py.txt", "inert_executed_verification_recipe")
for n in ("exact-links-stage1-design.html.txt", "exact-links-stage2-design.html.txt"):
    plan(PROOF / n, "rendered/" + n, "actual_progressive_href_check_HTML")
plan(PROOF / "candidate-rendered-report/design.html", "rendered/earlier-Markdown-design.html.txt", "actual_earlier_report_render")
plan(PROOF / "candidate-final-report/design.html", "rendered/final-Markdown-design.html.txt", "actual_final_report_render")
assert sha(raw(PROOF / "candidate-rendered-report/design.html")) == saved["proposed"]["actual_Markdown_builder_candidate"]["rendered_design_sha256"]
assert sha(raw(PROOF / "candidate-final-report/design.html")) == "972d6e26dfabf036376b09eedf3b5f71a3d711ed334659de6fb0148c56e196ea"
for n in ("references.json", "diagrams.json"):
    v = next(x for x in final["inputs"] if x["path"] == "publication/" + n)
    p = ROOT / v["path"]
    assert sha(raw(p)) == v["sha256"] and len(raw(p)) == v["bytes"]
    plan(p, "inputs/" + n, "actual_Markdown_render_input")
plan(Path(__file__), "recipes/assemble-publication-correction-archive.py.txt", "inert_archive_assembly_recipe")
files = sorted(p for p in SITE.rglob("*") if p.is_file())
assert len(files) == 53
canonical = []
for p in files:
    r = p.relative_to(SITE).as_posix()
    canonical.append(dict(ref(p), role="unchanged_frozen_A_site_dependency",
                          original_path="_site/" + r, site_relative_path=r))
for n in ("inputs.json", "readiness.json", "publication-link-disposition.json"):
    canonical.append(dict(ref(SITE.parent.parent / n), role="frozen_A_archive_provenance"))
aliases = []
def alias(path, digest, size=None):
    if any(x["original_recorded_path"] == path and x["sha256"] == digest for x in aliases):
        return
    match = next((x for x in copies if x["source_path"] == path and x["sha256"] == digest), None)
    if match is None:
        match = next((x for x in copies if x["sha256"] == digest), None)
    if match:
        if size is not None: assert match["bytes"] == size
        aliases.append(dict(original_recorded_path=path, sha256=digest, bytes=match["bytes"],
                            resolution="exact_archived_copy", resolved_path=match["archive_path"]))
        return
    match = next((x for x in canonical if x["sha256"] == digest), None)
    if match:
        if size is not None: assert match["bytes"] == size
        aliases.append(dict(original_recorded_path=path, sha256=digest, bytes=match["bytes"],
                            resolution="exact_canonical_frozen_A_reference", resolved_path=match["path"]))
        return
    if digest == "fc86a92fbd947df586cc1370d83d79b60910cec82c0c240bbfb68dc475dc88db":
        aliases.append(dict(original_recorded_path=path, sha256=digest, bytes=size,
                            resolution="historical_metadata_only_bytes_not_retained",
                            reason="Intermediate report read in initial receipt; never rendered. No source bytes claimed."))
        return
    raise AssertionError("Unresolved proof input: " + path + " " + digest)
def walk(v):
    if isinstance(v, dict):
        if isinstance(v.get("path"), str) and isinstance(v.get("sha256"), str):
            alias(v["path"], v["sha256"], v.get("bytes"))
        for x in v.values(): walk(x)
    elif isinstance(v, list):
        for x in v: walk(x)
for d in (focused, saved, final):
    walk(d)
for a in bindings["aliases"]:
    alias(a["original_recorded_path"], a["recorded_sha256"], a["bytes"])
    alias(a["snapshot_path"], a["recorded_sha256"], a["bytes"])
for c in canonical:
    if "original_path" not in c:
        continue
    alias(c["original_path"], c["sha256"], c["bytes"])
    for candidate in ("candidate-exact-links", "candidate-rendered-report", "candidate-final-report"):
        p = PROOF / candidate / c["site_relative_path"]
        b = raw(p)
        alias(rel(p), sha(b), len(b))
at = datetime.now(timezone.utc).isoformat()
readme = """# Publication link correction evidence

This separate archive preserves the known publication defect in source A (9e83fe7d41a52a63ba65b77730e392dafca8fffe) and the focused checks of proposed successor P. It does not change the frozen [A archive](../lab-9e83fe7/readiness.json), claim full acceptance of P, or claim remote publication is complete.

The original validator accepted existing repository files outside the served publication root. The proposed validator rejects resolved targets outside OUT.resolve(), including plain and percent-encoded escapes. The same five actual-function AST tests produced two expected escape failures and three passes on A, then five passes on the proposed guard. Both Ruff checks passed. Raw JUnits, logs (including four genuinely empty stderr files), exit records and the initial format-only correction remain byte-for-byte in proof/.

The saved original seven-page site is preserved by canonical references. The new validator rejects its two faulty design links progressively; changing exactly those two hrefs makes all seven candidate pages pass. The design report was also rendered with the actual Markdown-builder AST: first using report 4fad6d22..., then, after an explicit freeze, using final report 98b583b8.... All seven pages in the final candidate pass local validation. The final report differs from the earlier rendered report only in its status line; both corrected absolute GitHub URLs are unchanged. No full site/PDF rebuild, browser inspection, HTTP request, remote reachability check, or full suite was performed for this correction.

The initial correction receipt also recorded intermediate report hash fc86a92f... after the earlier render. That version was never rendered and its bytes were not retained. An attempted final-render preflight for it refused after the report changed. reference-aliases.json explicitly marks it metadata-only. It must not be interpreted as the current contents of its original mutable path.

sources/ contains the exact original/proposed builders, proposed tests and three retained report versions as inert text. rendered/ preserves the actual one-link, two-link, earlier Markdown and final Markdown HTML bytes as inert text, without rewriting relative links. The one-link snapshot intentionally still fails. The final rendering recipe was an inline execution recorded by the final receipt; it is not misrepresented as a separate retained script. The earlier actually executed verification script is retained under recipes/.

inputs.json binds every copied/generated archive file. reference-aliases.json resolves historical (path, SHA-256) pairs to these copies or the existing frozen A archive. canonical-references.json binds its 53 exact site files (including unchanged assets/PDFs) plus three provenance files; those dependencies are not duplicated. This directory is self-contained within the repository via explicit canonical references, not a standalone deployable site.

The five proposed tests are additional to A's 815-test campaign; they do not turn that campaign into an 820-test result. Later applied-repository checks, a successor commit identity, full CI/publication/Pages and final attestation must be recorded in separate additive evidence. Original receipts retain their then-current wording; this archive summary supplies their final scope.
"""
summary = dict(schema_version="1.0", created_at=at,
    status="FOCUSED_PROPOSED_CORRECTION_CHECKED_FULL_SUCCESSOR_ACCEPTANCE_PENDING",
    source_A_commit=A, successor="P proposed, no commit identity asserted here",
    frozen_A_archive_changed=False, copied_files=len(copies), copied_bytes=sum(x["bytes"] for x in copies),
    canonical_reference_files=len(canonical), focused_test_outcomes=suites,
    ruff_check_exit=0, ruff_format_exit=0,
    final_render_report_sha256=final["proposed_report"]["sha256"],
    final_render_design_sha256="972d6e26dfabf036376b09eedf3b5f71a3d711ed334659de6fb0148c56e196ea",
    all_seven_candidate_HTMLs_local_validation="PASS",
    intermediate_report_fc86="metadata-only, bytes not retained, never rendered",
    not_claimed=["Successor full acceptance", "Full site/PDF rebuild", "Browser review",
                 "Remote URL reachability", "CI/Pages/publication pass", "Goal completion"],
    no_live_tempfixtures_or_pyc_copied=True)
generated = {
    "README.md": readme.encode(),
    "summary.json": encode(summary),
    "canonical-references.json": encode(dict(schema_version="1.0", references=canonical,
        policy="Exact frozen A dependency bytes; no copies or HTML rewrites. Candidate design/report overrides are explicit in reference-aliases.json.")),
    "reference-aliases.json": encode(dict(schema_version="1.0", aliases=sorted(aliases, key=lambda x: (x["original_recorded_path"], x["sha256"])),
        unretained_version_is_not_evidence="fc86 report metadata only; final and earlier actual rendered inputs are archived."))}
for c in copies:
    p = ROOT / c["archive_path"]
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_bytes(raw(ROOT / c["source_path"]))
    assert sha(p.read_bytes()) == c["sha256"]
for n, b in generated.items():
    (OUT / n).write_bytes(b)
manifest = dict(schema_version="1.0", created_at=at, archive=rel(OUT), source_A_commit=A, copies=copies,
    generated=[dict(archive_path=rel(OUT / n), sha256=sha(b), bytes=len(b)) for n, b in generated.items()],
    canonical_references=dict(path=rel(OUT / "canonical-references.json"),
        sha256=sha(generated["canonical-references.json"]), bytes=len(generated["canonical-references.json"])),
    exclusions=["Live temporary fixtures", "__pycache__/pyc", "Duplicate PDF/assets trees",
                "Unretained intermediate fc86 report bytes", "Unrun successor full-acceptance claims"])
(OUT / "inputs.json").write_bytes(encode(manifest))
for p, b in cache.items():
    assert p.read_bytes() == b, "Source changed: " + str(p)
archive_files = sorted(p for p in OUT.rglob("*") if p.is_file())
receipt = dict(at=datetime.now(timezone.utc).isoformat(),
    status="ARCHIVE_BYTES_VERIFIED_NOT_SUCCESSOR_FULL_ACCEPTANCE", archive=rel(OUT),
    manifest=ref(OUT / "inputs.json"), copied_files=len(copies),
    generated_files_including_manifest=len(generated) + 1, all_archive_files=len(archive_files),
    all_archive_bytes=sum(p.stat().st_size for p in archive_files),
    canonical_reference_count=len(canonical), all_copied_bytes_match=True, all_read_inputs_unchanged=True,
    exact_zero_byte_logs_retained=[x["archive_path"] for x in copies if x["bytes"] == 0],
    pending=["Parent review", "Public payload extension review", "Source application/commit",
             "Actual-repository additive verification", "Successor full remote gates"])
receipt_path = PROOF / "archive-assembly-receipt.json"
assert not receipt_path.exists()
receipt_path.write_bytes(encode(receipt))
print(json.dumps(receipt, indent=2))
