"""Sparse read-only evidence capture for one frozen acceptance invocation."""
from __future__ import annotations

import argparse
import datetime
import hashlib
import json
import pathlib
import re
import sqlite3
import time
import xml.etree.ElementTree as ET
from contextlib import closing

ROOT = pathlib.Path(__file__).resolve().parents[3]
OUT: pathlib.Path
ACCEPTANCE: pathlib.Path
COMMIT: str
PASS1_ROOT: str
PASS2_ROOT: str | None
MIN_STAMP: str
CASES = {"happy": "", "DROP_ACK_AFTER_EFFECT": "DROP_ACK_AFTER_EFFECT", "DUPLICATE_DELIVERY": "DUPLICATE_DELIVERY", "WMS_UNAVAILABLE": "WMS_UNAVAILABLE"}
PATTERNS = {
    "credential_uri": re.compile(r"\b(?:postgres(?:ql)?|amqps?|https?)://[^\s\"'<>/@]+:[^\s\"'<>/@]+@", re.I),
    "private_key": re.compile(r"-----BEGIN (?:RSA |EC |OPENSSH )?PRIVATE KEY-----"),
    "bearer_basic": re.compile(r"\b(?:Bearer|Basic)\s+[A-Za-z0-9_+/.=-]{12,}"),
    "provider_key": re.compile(r"\b(?:gh[pousr]_[A-Za-z0-9]{20,}|github_pat_[A-Za-z0-9_]{20,}|sk-[A-Za-z0-9_-]{20,}|AKIA[A-Z0-9]{16})\b"),
    "secret_assignment": re.compile(r'''["'](?:password|passwd|client_secret|api_key|access_token|refresh_token)["']\s*:\s*["']([^"']+)["']''', re.I),
}


def now():
    return datetime.datetime.now(datetime.UTC).isoformat()


def read(path):
    return json.loads(path.read_text(encoding="utf-8-sig"))


def write(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".tmp")
    temporary.write_text(json.dumps(value, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
    temporary.replace(path)


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def query_journals(directory):
    result = {}
    for name in ("runtime.db", "plc.db"):
        path = directory / name
        with closing(sqlite3.connect(path.resolve().as_uri() + "?mode=ro", uri=True, timeout=2)) as connection, connection:
            connection.execute("PRAGMA query_only=ON")
            connection.execute("BEGIN")
            if name == "runtime.db":
                rows = connection.execute("SELECT id,payload_hash,receipt FROM controller_journal").fetchall()
                facts = {"controller_journal": [{"command_id": row[0], "payload_hash": row[1], "receipt": json.loads(row[2])} for row in rows], "robot_event_count": connection.execute("SELECT COUNT(*) FROM robot_events").fetchone()[0]}
            else:
                rows = connection.execute("SELECT command_id,payload_hash,state,result_sequence,acknowledged,result FROM plc_commands").fetchall()
                results = connection.execute("SELECT command_id,sequence,receipt FROM plc_results").fetchall()
                facts = {"plc_commands": [{"command_id": row[0], "payload_hash": row[1], "state": row[2], "result_sequence": row[3], "acknowledged": bool(row[4]), "result": json.loads(row[5]) if row[5] else None} for row in rows], "plc_results": [{"command_id": row[0], "sequence": row[1], "receipt": json.loads(row[2])} for row in results]}
        result[name] = {"path": str(path), "database_sha256_at_capture": sha(path), "read_only": True, **facts}
    return result


def capture(stack_path, pass_number, manifest):
    directory = stack_path.parent
    name = directory.name
    stack = read(stack_path)
    session_path = directory / "session.json"
    if not session_path.exists():
        session_path = directory / "last-session.json"
    session = read(session_path)
    # Preserve temporary journals first, before doing optional local extraction.
    journals = query_journals(pathlib.Path(stack["data_directory"]))
    destination = OUT / f"pass-{pass_number}" / f"{name.lower()}.json"
    data = {
        "captured_at_utc": now(), "capture_complete": False, "acceptance_pass": pass_number,
        "acceptance_directory": ACCEPTANCE.relative_to(ROOT).as_posix(),
        "source": manifest["source"], "artifact_directory": directory.relative_to(ROOT).as_posix(),
        "formal_test_status": "PENDING_JUNIT", "runtime": stack["runtime"],
        "session": {key: session[key] for key in ("session_id", "correlation_id", "order_id", "job_id", "command_id", "status", "current_stage", "revision", "updated_at")},
        "journals": journals, "stack": stack,
        "physical_effect_count": sum(item["receipt"]["effect_count"] for item in journals["runtime.db"]["controller_journal"]),
    }
    write(destination, data)
    browser = read(directory / "browser.json")
    hashes = read(directory / "source-hashes.json")
    source_checks = []
    for filename, recorded in hashes.items():
        source_path = pathlib.Path(filename)
        if not source_path.is_absolute():
            source_path = ROOT / filename
        source_checks.append({"path": source_path.relative_to(ROOT).as_posix(), "recorded_sha256": recorded, "matches_current_file": sha(source_path) == recorded})
    steps = session["steps"]
    data.update({
        "source_checks": source_checks, "all_recorded_source_files_match": all(item["matches_current_file"] for item in source_checks),
        "browser_errors": browser["errors"], "browser_console_errors": browser["console_errors"],
        "transitions": [{key: step[key] for key in ("stage", "sequence", "status", "summary", "state_before", "state_after", "classification", "protocol")} for step in steps],
        "important_outputs": [{"stage": step["stage"], "sequence": step["sequence"], "status": step["status"], "classification": step["classification"], "output": step["output"]} for step in steps if step["stage"] in (10, 11, 12, 16, 17, 20, 21, 22)],
        "verification": [{"sequence": step["sequence"], "state_before": step["state_before"], "state_after": step["state_after"], "output": step["output"].get("verification", step["output"])} for step in steps if step["stage"] == 19],
        "input_files": [{"path": path.relative_to(ROOT).as_posix(), "sha256": sha(path)} for path in (stack_path, session_path, directory / "browser.json", directory / "source-hashes.json")],
    })
    if (directory / "blender-motion.json").exists():
        data["motion"] = read(directory / "blender-motion.json")
        data["motion"]["rendered_position_changed"] = data["motion"]["before"]["rendered"] != data["motion"]["after"]["rendered"]
    findings = []
    for path in [directory / "browser.json", session_path, stack_path, *directory.glob("*.log")]:
        text = path.read_text(encoding="utf-8-sig", errors="replace")
        for label, pattern in PATTERNS.items():
            count = len(list(pattern.finditer(text)))
            if count:
                findings.append({"path": path.relative_to(ROOT).as_posix(), "pattern": label, "count": count, "values": "withheld"})
    data["secret_review"] = {"potential_findings": findings, "scope": "Stored case/session/browser/stack JSON and service logs only. Trace ZIP review is deferred until the full suites finish to avoid competing workload. No guarantee of exhaustive secret absence."}
    data["capture_complete"] = True
    write(destination, data)
    print(f"{now()} CAPTURE pass={pass_number} case={name} state={session['status']} stage={session['current_stage']} effects={data['physical_effect_count']}", flush=True)
    return destination


def refresh_junit(pass_number):
    junit_path = ACCEPTANCE / f"tests-{pass_number}.xml"
    if not junit_path.exists():
        return {"status": "PENDING_JUNIT"}
    try:
        suite = ET.parse(junit_path).getroot().find("testsuite")
    except ET.ParseError:
        return {"status": "PENDING_JUNIT_WRITE"}
    counts = {name: int(suite.attrib.get(name, "0")) for name in ("tests", "failures", "errors", "skipped")}
    counts["passed"] = counts["tests"] - counts["failures"] - counts["errors"] - counts["skipped"]
    formal = {"status": "FAILED" if counts["failures"] or counts["errors"] else "PASSED", "junit": junit_path.relative_to(ROOT).as_posix(), "seconds": float(suite.attrib["time"]), **counts}
    for path in (OUT / f"pass-{pass_number}").glob("*.json"):
        data = read(path)
        scenario = CASES[path.stem.upper() if path.stem != "happy" else "happy"]
        expected_name = f"test_full_lab_browser_authorization_recovery_and_business_completion[{scenario}]"
        case = next((case for case in suite if case.attrib.get("classname") == "tests.lab.test_browser_lab" and case.attrib.get("name") == expected_name), None)
        if case is None:
            continue
        result = "FAILED" if case.find("failure") is not None or case.find("error") is not None else "SKIPPED" if case.find("skipped") is not None else "PASSED"
        if data.get("formal_test_status") != result:
            data["formal_test_status"] = result
            data["formal_test_result"] = {"junit": junit_path.relative_to(ROOT).as_posix(), "name": expected_name, "seconds": float(case.attrib["time"])}
            write(path, data)
    return formal


def poll():
    global PASS2_ROOT
    manifest = read(ACCEPTANCE / "manifest.json")
    if manifest["source"]["commit"] != COMMIT:
        raise RuntimeError("Acceptance source identity differs from this watcher scope")
    errors = []
    stack_paths = []
    for stamp in (ROOT / "artifacts/lab-browser").iterdir():
        if not re.fullmatch(r"\d{8}T\d{12}", stamp.name) or stamp.name < MIN_STAMP:
            continue
        for case_name in CASES:
            stack_path = stamp / case_name / "stack.json"
            if stack_path.exists():
                stack_paths.append(stack_path)
    for stack_path in sorted(stack_paths):
        stack = read(stack_path)
        match = re.search(r"robotops-acceptance-([12])-[^\\/]+", stack["data_directory"])
        if not match:
            continue
        pass_number = int(match.group(1))
        if pass_number == 1 and match.group(0) != PASS1_ROOT:
            continue
        if pass_number == 2:
            if PASS2_ROOT is None:
                PASS2_ROOT = match.group(0)
                print(f"{now()} bound pass2 fixture root={PASS2_ROOT}", flush=True)
            elif match.group(0) != PASS2_ROOT:
                continue
        if not (stack_path.parent / "browser.json").exists():
            continue
        destination = OUT / f"pass-{pass_number}" / f"{stack_path.parent.name.lower()}.json"
        if destination.exists() and read(destination).get("capture_complete"):
            continue
        try:
            capture(stack_path, pass_number, manifest)
        except Exception as error:
            errors.append({"artifact": stack_path.parent.relative_to(ROOT).as_posix(), "error": f"{type(error).__name__}: {error}"})
    passes = {}
    for number in (1, 2):
        formal = refresh_junit(number)
        saved = [read(path) for path in (OUT / f"pass-{number}").glob("*.json")]
        passes[str(number)] = {"suite": formal, "captured_cases": [{"path": f"pass-{number}/{pathlib.Path(data['artifact_directory']).name.lower()}.json", "case": pathlib.Path(data["artifact_directory"]).name, "session": data["session"], "physical_effect_count": data["physical_effect_count"], "capture_complete": data["capture_complete"], "formal_test_status": data["formal_test_status"]} for data in saved]}
    status = {"polled_at_utc": now(), "watcher_status": "OBSERVING", "completion": "NOT ACHIEVED: this pack records browser/journal facts, not whole-goal completion", "acceptance_directory": ACCEPTANCE.relative_to(ROOT).as_posix(), "source": manifest["source"], "fixture_roots": {"1": PASS1_ROOT, "2": PASS2_ROOT}, "poll_interval_seconds": 60, "passes": passes, "capture_errors": errors, "scope": "Read-only artifact and temporary SQLite inspection. No tests, browsers, Blender processes or source changes. Case results require terminal JUnit; trace ZIP review deferred."}
    write(OUT / "status.json", status)
    return all(len(value["captured_cases"]) == 4 and all(case["capture_complete"] for case in value["captured_cases"]) and value["suite"]["status"] in ("PASSED", "FAILED") for value in passes.values())


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--acceptance-dir", required=True, type=pathlib.Path)
    parser.add_argument("--expected-commit", required=True)
    parser.add_argument("--pass1-root", required=True)
    parser.add_argument("--pass2-root")
    parser.add_argument("--output-dir", required=True, type=pathlib.Path)
    parser.add_argument("--watch", action="store_true")
    args = parser.parse_args()
    ACCEPTANCE = args.acceptance_dir.resolve(strict=True)
    OUT = args.output_dir.resolve()
    if not ACCEPTANCE.is_relative_to(ROOT / "docs/evidence/acceptance"):
        parser.error("--acceptance-dir must name this repository's acceptance evidence directory")
    if not OUT.is_relative_to(ROOT / "artifacts"):
        parser.error("--output-dir must stay in this repository's ignored artifacts directory")
    manifest = read(ACCEPTANCE / "manifest.json")
    COMMIT = args.expected_commit
    if manifest["source"]["commit"] != COMMIT:
        parser.error("--expected-commit does not match the acceptance manifest")
    PASS1_ROOT = args.pass1_root
    PASS2_ROOT = args.pass2_root
    if not re.fullmatch(r"robotops-acceptance-1-[^\\/]+", PASS1_ROOT):
        parser.error("--pass1-root must be the exact fixture-root basename")
    if PASS2_ROOT and not re.fullmatch(r"robotops-acceptance-2-[^\\/]+", PASS2_ROOT):
        parser.error("--pass2-root must be the exact fixture-root basename")
    MIN_STAMP = datetime.datetime.fromisoformat(manifest["started_at"].replace("Z", "+00:00")).astimezone(datetime.UTC).strftime("%Y%m%dT%H%M%S")
    if (OUT / "status.json").exists():
        saved_status = read(OUT / "status.json")
        if saved_status["source"]["commit"] != COMMIT or saved_status["acceptance_directory"] != ACCEPTANCE.relative_to(ROOT).as_posix():
            parser.error("--output-dir already belongs to another source or acceptance; choose a fresh directory")
        PASS2_ROOT = PASS2_ROOT or saved_status.get("fixture_roots", {}).get("2")
    started = time.monotonic()
    print(f"{now()} watcher started scope={COMMIT} once_per_minute={args.watch}", flush=True)
    while True:
        interruption_path = ACCEPTANCE / "interruption.json"
        if interruption_path.exists():
            interruption = read(interruption_path)
            status = read(OUT / "status.json") if (OUT / "status.json").exists() else {"source": manifest["source"], "acceptance_directory": ACCEPTANCE.relative_to(ROOT).as_posix(), "passes": {}}
            status.update({"watcher_status": "STOPPED_INTERRUPTED", "stopped_at_utc": now(), "completion": "NOT ACHIEVED: acceptance interrupted; no whole-suite pass inferred", "interruption": interruption, "interruption_evidence": interruption_path.relative_to(ROOT).as_posix()})
            for number, data in status["passes"].items():
                if data["suite"]["status"].startswith("PENDING"):
                    data["suite"] = {"status": "INTERRUPTED_OR_NOT_RUN", "meaning": "No terminal JUnit; consult interruption evidence"}
            write(OUT / "status.json", status)
            print(f"{now()} watcher stopped because acceptance interruption marker exists", flush=True)
            break
        done = poll()
        if done or not args.watch:
            status = read(OUT / "status.json")
            status.update({"watcher_status": "CAPTURE_COMPLETE" if done else "CHECK_COMPLETE", "stopped_at_utc": now()})
            write(OUT / "status.json", status)
            print(f"{now()} watcher stopped captures_and_junit_complete={done}", flush=True)
            break
        if time.monotonic() - started > 3 * 60 * 60:
            status = read(OUT / "status.json")
            status.update({"watcher_status": "STOPPED_OBSERVATION_LIMIT", "stopped_at_utc": now()})
            write(OUT / "status.json", status)
            print(f"{now()} watcher stopped at three-hour observation limit; inspect status.json", flush=True)
            break
        time.sleep(60)
