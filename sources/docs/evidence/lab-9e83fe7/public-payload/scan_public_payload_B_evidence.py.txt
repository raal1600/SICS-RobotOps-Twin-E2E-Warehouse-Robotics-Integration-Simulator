def relative(path):
    return str(path.relative_to(ROOT)).replace("\\", "/")


def load_bound(path):
    data = path.read_bytes()
    return json.loads(data), {"path": relative(path), "bytes": len(data), "sha256": m.sha(data)}


inputs = []
for path, expected in EXPECTED.items():
    data = (ROOT / path).read_bytes()

        raise RuntimeError("Frozen input differs: " + path)
    inputs.append({"path": path, "sha256": expected, "bytes": len(data)})

url_regex = re.compile(r"\b(?:https?|amqps?|postgres(?:ql)?|mongodb(?:\+srv)?)://[^\s\"'<>]+", re.I)


def scan_text(text, location, pointer="", structured=True):
    global url_occurrences
    for match in url_regex.finditer(text):
        url_occurrences += 1
        try:
            parsed = urlsplit(match.group())
            if parsed.password is not None:
                m.candidate("uri_decoded_userinfo", unquote(parsed.username or "") + ":" + unquote(parsed.password), location, pointer + "/url-char:" + str(match.start()))
            for key, value in parse_qsl(parsed.query, keep_blank_values=True):
                normalized = re.sub(r"[^a-z0-9]", "", key.lower())
                if normalized in m.SENSITIVE or normalized in {"accesskey", "credential", "credentials"}:
                    m.candidate("uri_query:" + normalized, value, location, pointer + "/url-char:" + str(match.start()))
        except ValueError as exc:
            url_errors.append({"location": location, "pointer": pointer + "/url-char:" + str(match.start()), "uri_sha256": m.sha(match.group().encode()), "error": str(exc)})
    original_scan(text, location, pointer, structured)


m.scan_text = scan_text
