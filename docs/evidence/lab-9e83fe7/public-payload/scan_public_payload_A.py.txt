from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
OUT = Path(__file__).resolve().parent
A = "9e83fe7d41a52a63ba65b77730e392dafca8fffe"
BASE = "814979d600180246b6b89233703e99fb4f085e44"
STARTED = dt.datetime.now(dt.UTC).isoformat()
SENSITIVE = {
    "password", "passwd", "pwd", "secret", "token", "authtoken", "idtoken",
    "xauthtoken", "xcsrftoken", "clientsecret", "accesstoken", "refreshtoken",
    "authorization", "proxyauthorization", "cookie", "setcookie", "apikey",
    "xapikey", "secretkey", "privatekey", "awssecretaccesskey", "sessiontoken",
}
PATTERNS = [
    ("private_key", re.compile(r"-----BEGIN (?:RSA |EC |OPENSSH |DSA )?PRIVATE KEY-----")),
    ("credential_prefix", re.compile(r"\b(?:AKIA[0-9A-Z]{16}|gh[pousr]_[A-Za-z0-9]{20,}|github_pat_[A-Za-z0-9_]{30,}|sk-(?:proj-)?[A-Za-z0-9_-]{24,}|xox[baprs]-[A-Za-z0-9-]{15,})\b")),
    ("jwt_shape", re.compile(r"\beyJ[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}\b")),
    ("authorization_literal", re.compile(r"\b(?:Bearer|Basic)\s+([A-Za-z0-9_./+=:-]{8,})", re.I)),
    ("credential_assignment", re.compile(r"[\"']?(password|passwd|client_secret|access_token|refresh_token|api[_-]?key|aws_secret_access_key)[\"']?\s*[:=]\s*[\"']([^\"'\r\n]{1,512})[\"']", re.I)),
    ("unquoted_credential_assignment", re.compile(r"\b(password|passwd|pwd|client_secret|access_token|refresh_token|api[_-]?key|aws_secret_access_key|Authorization|Cookie|Set-Cookie)\s*[:=]\s*([^\s,;\"'<>]{3,512})", re.I)),
    ("url_userinfo", re.compile(r"\b(?:https?|amqps?|postgres(?:ql)?|mongodb(?:\+srv)?)://([^\s/@]+:[^\s/@]+)@", re.I)),
]
counts = collections.Counter()
errors = []
json_errors = []
candidate_groups = {}
zip_details = []
db_details = []
pdf_details = []
embedded_depth_limits = []
scan_results = {}


def sha(data):
    return hashlib.sha256(data).hexdigest()


def git(*args, data=None):
    return subprocess.check_output(["git", *args], cwd=ROOT, input=data)


def tree(commit):
    result = {}
    for item in git("ls-tree", "-r", "-z", "--long", commit).split(b"\0"):
        if not item:
            continue
        meta, path = item.split(b"\t", 1)
        mode, typ, oid, size = meta.decode().split()
        result[path.decode("utf-8")] = {
            "mode": mode, "type": typ, "git_oid": oid, "bytes": int(size)
        }
    return result


def classification(value):
    v = value.strip()
    if v in {"demo-only", "guest:guest", "robotops:demo-only"}:
        return "KNOWN_PUBLIC_DEMO_CREDENTIAL"
    if v.casefold() in {"[redacted]", "<redacted>", "***", "redacted", "null", "none", "undefined", "false", "true"}:
        return "REDACTION_OR_NONCREDENTIAL_LITERAL"
    return "CONTEXT_REVIEW_REQUIRED"


def candidate(category, value, location, pointer):
    if not isinstance(value, str) or not value.strip():
        return
    if len(value) > 4096:
        # Avoid treating entire source excerpts under generic 'token' keys as secrets;
        # their complete strings are still independently pattern scanned.
        counts["oversize_sensitive_scalar_pattern_scanned"] += 1
        scan_text(value, location, pointer + "/oversize", structured=False)
        return
    key = (category, sha(value.encode("utf-8")), classification(value))
    group = candidate_groups.setdefault(key, {
        "category": category, "value_sha256": key[1], "classification": key[2],
        "value_length": len(value), "value": "[REDACTED BY AUDIT]",
        "occurrences": [],
    })
    group["occurrences"].append({"location": location, "pointer": pointer})


def walk(obj, location, pointer="", depth=0):
    if isinstance(obj, dict):
        header_name = obj.get("name")
        if isinstance(header_name, str):
            norm = re.sub(r"[^a-z0-9]", "", header_name.lower())
            if norm in SENSITIVE and isinstance(obj.get("value"), str):
                candidate("structured_header:" + norm, obj["value"], location, pointer + "/value")
        for key, value in obj.items():
            p = pointer + "/" + str(key).replace("~", "~0").replace("/", "~1")
            norm = re.sub(r"[^a-z0-9]", "", str(key).lower())
            if norm in SENSITIVE and isinstance(value, str):
                candidate("structured_key:" + norm, value, location, p)
            walk(value, location, p, depth)
    elif isinstance(obj, list):
        for i, value in enumerate(obj):
            walk(value, location, pointer + "/" + str(i), depth)
    elif isinstance(obj, str):
        low = obj.lower()
        if any(w in low for w in ("password", "passwd", "secret", "token", "authorization", "cookie", "private key", "bearer", "://", "api_key", "apikey")):
            scan_text(obj, location, pointer + "/decoded-string", structured=False)
        stripped = obj.strip()
        if stripped.startswith(("{", "[")) and stripped.endswith(("}", "]")):
            if depth >= 8:
                embedded_depth_limits.append({"location": location, "pointer": pointer})
                return
            try:
                nested = json.loads(stripped)
            except (ValueError, RecursionError):
                return
            counts["embedded_json_parses"] += 1
            walk(nested, location, pointer + "/embedded", depth + 1)


def scan_text(text, location, pointer="", structured=True):
    counts["decoded_text_characters_scanned"] += len(text)
    for category, regex in PATTERNS:
        for match in regex.finditer(text):
            if category in {"credential_assignment", "unquoted_credential_assignment"}:
                value = match.group(2)
            elif match.lastindex:
                value = match.group(match.lastindex)
            else:
                value = match.group(0)
            candidate(category, value, location, pointer + "/char:" + str(match.start()))
    if not structured:
        return
    ext = Path(location.split("!")[-1]).suffix.lower()
    stripped = text.strip()
    if ext == ".json":
        try:
            walk(json.loads(text), location)
            counts["json_documents_parsed"] += 1
        except (ValueError, RecursionError) as exc:
            json_errors.append({"location": location, "error": str(exc)})
    elif ext in {".jsonl", ".trace", ".network"}:
        for i, line in enumerate(text.splitlines()):
            if not line.strip():
                continue
            try:
                walk(json.loads(line), location, "/line:" + str(i + 1))
                counts["json_lines_parsed"] += 1
            except (ValueError, RecursionError) as exc:
                json_errors.append({"location": location, "line": i + 1, "error": str(exc)})
    elif stripped.startswith(("{", "[")) and stripped.endswith(("}", "]")):
        try:
            walk(json.loads(stripped), location)
            counts["content_detected_json_documents_parsed"] += 1
        except (ValueError, RecursionError):
            pass


def database(data, location, oid):
    tempdir = OUT / "immutable-A-blobs"
    tempdir.mkdir(exist_ok=True)
    tmp = tempdir / (oid + ".db")
    if tmp.exists():
        raise RuntimeError("Refusing existing immutable DB copy: " + str(tmp))
    tmp.write_bytes(data)
    record = {"location": location, "git_oid": oid, "sha256": sha(data), "bytes": len(data), "tables": []}
    try:
        connection = sqlite3.connect(tmp.as_uri() + "?mode=ro&immutable=1", uri=True)
        try:
            record["integrity_check"] = [r[0] for r in connection.execute("PRAGMA integrity_check")]
            schema = connection.execute("SELECT type,name,tbl_name,rootpage,sql FROM sqlite_master ORDER BY type,name").fetchall()
            schema_data = json.dumps(schema, ensure_ascii=False, separators=(",", ":"))
            record["schema_sha256"] = sha(schema_data.encode())
            scan_text(schema_data, location + "!sqlite-schema", structured=False)
            for typ, name, *_ in schema:
                if typ != "table":
                    continue
                table = {"name": name, "rows": 0, "text_values": 0, "blob_values": 0}
                digest = hashlib.sha256()
                quoted = '"' + name.replace('"', '""') + '"'
                cursor = connection.execute("SELECT * FROM " + quoted)
                table["columns"] = [c[0] for c in cursor.description]
                for row_index, row in enumerate(cursor):
                    typed = []
                    for col_index, value in enumerate(row):
                        loc = location + "!sqlite/" + name + "/" + str(row_index) + "/" + table["columns"][col_index]
                        if isinstance(value, bytes):
                            table["blob_values"] += 1
                            inspect_bytes(value, loc, oid + "-" + str(row_index) + "-" + str(col_index), allow_db=False)
                            typed.append({"type": "blob", "length": len(value), "sha256": sha(value)})
                        else:
                            typed.append({"type": type(value).__name__, "value": value})
                            if isinstance(value, str):
                                table["text_values"] += 1
                                scan_text(value, loc)
                                norm = re.sub(r"[^a-z0-9]", "", table["columns"][col_index].lower())
                                if norm in SENSITIVE:
                                    candidate("sqlite_column:" + norm, value, loc, "")
                    digest.update(json.dumps(typed, ensure_ascii=False, separators=(",", ":")).encode())
                    digest.update(b"\n")
                    table["rows"] += 1
                table["ordered_typed_rows_sha256"] = digest.hexdigest()
                record["tables"].append(table)
        finally:
            connection.close()
        record["copy_sha256_after_read"] = sha(tmp.read_bytes())
        if record["copy_sha256_after_read"] != record["sha256"]:
            raise RuntimeError("Immutable DB copy changed")
    finally:
        tmp.unlink()
        if not any(tempdir.iterdir()):
            tempdir.rmdir()
    db_details.append(record)
    counts["sqlite_files_extracted"] += 1
    return {"kind": "sqlite_all_tables_typed_values_extracted", "table_count": len(record["tables"])}


def pdf(data, location):
    from pypdf import PdfReader
    reader = PdfReader(io.BytesIO(data))
    record = {"location": location, "sha256": sha(data), "pages": [], "metadata_scanned": True}
    scan_text(str(reader.metadata), location + "!pdf-metadata", structured=False)
    for i, page in enumerate(reader.pages):
        text = page.extract_text() or ""
        scan_text(text, location + "!pdf-page-" + str(i + 1), structured=False)
        record["pages"].append({"page": i + 1, "text_characters": len(text), "text_sha256": sha(text.encode())})
    pdf_details.append(record)
    counts["pdf_files_extracted"] += 1
    counts["pdf_pages_extracted"] += len(record["pages"])
    return {"kind": "pdf_all_page_text_extracted", "pages": len(record["pages"])}


def inspect_bytes(data, location, oid, allow_db=True, depth=0):
    counts["full_content_read_bytes_including_members"] += len(data)
    if data.startswith(b"PK\x03\x04") or zipfile.is_zipfile(io.BytesIO(data)):
        record = {"location": location, "sha256": sha(data), "compressed_bytes": len(data), "members": []}
        with zipfile.ZipFile(io.BytesIO(data)) as archive:
            for i, info in enumerate(archive.infolist()):
                raw = archive.read(info)
                if len(raw) != info.file_size:
                    raise RuntimeError("ZIP size mismatch")
                member_location = location + "!" + str(i) + ":" + info.filename
                member = {
                    "index": i, "name": info.filename, "bytes": len(raw),
                    "compressed_bytes": info.compress_size, "crc32": format(info.CRC, "08x"),
                    "crc_checked": True, "sha256": sha(raw),
                }
                member["disposition"] = inspect_bytes(raw, member_location, oid + "-" + str(i), depth=depth + 1)
                record["members"].append(member)
                counts["zip_members_fully_read"] += 1
                counts["zip_member_uncompressed_bytes"] += len(raw)
        zip_details.append(record)
        counts["zip_archives_fully_read"] += 1
        return {"kind": "zip_all_members_full_read_crc_hash_scan", "members": len(record["members"])}
    if data.startswith(b"SQLite format 3\x00") and allow_db:
        return database(data, location, oid)
    if data.startswith(b"%PDF"):
        # Whole raw PDF bytes also receive literal pattern scan; page streams use extraction.
        scan_text(data.decode("latin-1"), location + "!raw-pdf-bytes", structured=False)
        return pdf(data, location)
    if not data:
        return {"kind": "empty"}
    if data.startswith((b"\xff\xfe", b"\xfe\xff")):
        text = data.decode("utf-16")
        encoding = "utf-16-bom"
    else:
        try:
            text = data.decode("utf-8-sig")
            encoding = "utf-8-sig"
        except UnicodeDecodeError:
            if b"\0" not in data and Path(location.split("!")[-1]).suffix.lower() not in {".png", ".jpg", ".jpeg", ".gif", ".webp", ".ico", ".woff", ".woff2", ".ttf"}:
                text = data.decode("latin-1")
                encoding = "latin-1-lossless-fallback"
            else:
                # No semantic/image OCR claim. Still check all raw bytes for ASCII literals.
                scan_text(data.decode("latin-1"), location + "!raw-binary-bytes", structured=False)
                counts["binary_files_literal_scanned"] += 1
                return {"kind": "binary_full_hash_and_raw_literal_scan_no_ocr", "signature_hex": data[:16].hex()}
    if "\0" in text:
        scan_text(data.decode("latin-1"), location + "!raw-binary-bytes", structured=False)
        counts["binary_files_literal_scanned"] += 1
        return {"kind": "binary_full_hash_and_raw_literal_scan_no_ocr", "signature_hex": data[:16].hex()}
    scan_text(text, location)
    counts["text_files_fully_scanned"] += 1
    counts["text_file_decoded_bytes"] += len(data)
    return {"kind": "full_text_pattern_and_structured_scan", "encoding": encoding}


def dump(name, value):
    path = OUT / name
    if path.exists():
        raise RuntimeError("Refusing overwrite: " + str(path))
    path.write_text(json.dumps(value, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
    return {"path": str(path.relative_to(ROOT)).replace("\\", "/"), "sha256": sha(path.read_bytes()), "bytes": path.stat().st_size}


def main():
    if ROOT.name != "robotops-twin":
        raise RuntimeError("Wrong repository root: " + str(ROOT))
    actual_head = git("rev-parse", "HEAD").decode().strip()
    # HEAD may acquire evidence successor later; this program reads only explicitly pinned A.
    a_tree, base_tree = tree(A), tree(BASE)
    commits = git("rev-list", A, "^" + BASE).decode().splitlines()
    new_objects_raw = git("rev-list", "--objects", A, "^" + BASE)
    object_paths = {}
    for line in new_objects_raw.decode().splitlines():
        oid, _, path = line.partition(" ")
        object_paths[oid] = path
    meta_raw = git("cat-file", "--batch-check=%(objectname) %(objecttype) %(objectsize)", data=("\n".join(object_paths) + "\n").encode())
    objects = []
    new_blobs = {}
    for line in meta_raw.decode().splitlines():
        oid, typ, size = line.split()
        entry = {"git_oid": oid, "type": typ, "bytes": int(size), "representative_path": object_paths[oid]}
        objects.append(entry)
        if typ == "blob":
            new_blobs[oid] = entry
    occurrences = collections.defaultdict(list)
    for commit in commits:
        for path, entry in tree(commit).items():
            oid = entry["git_oid"]
            if oid in new_blobs:
                occurrences[oid].append({"commit": commit, "path": path})
    a_oids = {v["git_oid"] for v in a_tree.values()}
    base_oids = {v["git_oid"] for v in base_tree.values()}
    aliases = collections.defaultdict(list)
    for path, entry in a_tree.items():
        aliases[entry["git_oid"]].append(path)
    scan_order = list(aliases)
    scan_order += [oid for oid in new_blobs if oid not in a_oids]
    batch = subprocess.Popen(["git", "cat-file", "--batch"], cwd=ROOT, stdin=subprocess.PIPE, stdout=subprocess.PIPE)
    try:
        for index, oid in enumerate(scan_order):
            batch.stdin.write((oid + "\n").encode())
            batch.stdin.flush()
            header = batch.stdout.readline().decode().strip().split()
            if len(header) != 3 or header[0] != oid or header[1] != "blob":
                raise RuntimeError("Unexpected git blob header")
            size = int(header[2])
            raw = batch.stdout.read(size)
            if len(raw) != size or batch.stdout.read(1) != b"\n":
                raise RuntimeError("Truncated git blob")
            git_hash = hashlib.sha1(b"blob " + str(size).encode() + b"\0" + raw).hexdigest()
            if git_hash != oid:
                raise RuntimeError("Git object content verification failed")
            paths = aliases.get(oid) or sorted({x["path"] for x in occurrences[oid]})
            location = paths[0] if oid in a_oids else "history/" + oid + "/" + paths[0]
            disposition = inspect_bytes(raw, location, oid)
            scan_results[oid] = {
                "git_oid": oid, "sha256": sha(raw), "bytes": size,
                "git_blob_hash_verified": True, "scan_location": location,
                "A_tree_paths": aliases.get(oid, []), "disposition": disposition,
                "new_reachable_from_base": oid in new_blobs,
            }
            if index % 100 == 0:
                print(json.dumps({"scanned_unique_blobs": index + 1, "total": len(scan_order), "candidate_groups": len(candidate_groups)}), flush=True)
    finally:
        batch.stdin.close()
        batch.stdout.close()
        batch.wait(timeout=20)
    files = []
    for path, entry in a_tree.items():
        previous = base_tree.get(path)
        state = "added" if previous is None else ("unchanged_already_published" if previous == entry else "modified")
        files.append({"path": path, **entry, "baseline_comparison": state, "baseline": previous, "scan": scan_results[entry["git_oid"]]})
    history = []
    for oid, entry in new_blobs.items():
        history.append({
            **entry, "occurrences_in_new_commit_trees": occurrences[oid],
            "present_in_A_tree": oid in a_oids,
            "review_scope": "exact_blob_reuse_from_full_A_scan" if oid in a_oids else "superseded_blob_full_scan",
            "scan": scan_results[oid],
        })
    finished = dt.datetime.now(dt.UTC).isoformat()
    common = {
        "source_A": A, "source_A_tree": git("rev-parse", A + "^{tree}").decode().strip(),
        "baseline": BASE, "baseline_tree": git("rev-parse", BASE + "^{tree}").decode().strip(),
        "head_observed_at_start": actual_head, "started_utc": STARTED, "finished_utc": finished,
        "scanner_sha256": sha(Path(__file__).read_bytes()),
        "method": "Exact git blobs; verified Git SHA1 + SHA256; full text and decoded structured strings; full ZIP member CRC/read/hash; immutable SQLite all table values; PDF all page text; no prefix sampling; no prior scan reuse.",
    }
    inventory = {
        **common, "status": "FULL_CONTENT_SCAN_COMPLETE_CONTEXT_REVIEW_PENDING",
        "files": files, "removed_baseline_paths": sorted(set(base_tree) - set(a_tree)),
        "totals": {"A_paths": len(files), "A_path_bytes": sum(x["bytes"] for x in files),
                   "A_unique_blobs": len(a_oids), "baseline_states": dict(collections.Counter(x["baseline_comparison"] for x in files)),
                   "all_scanned_unique_blobs_including_history": len(scan_results),
                   "all_scanned_unique_blob_bytes": sum(x["bytes"] for x in scan_results.values()),
                   "history_only_unique_blobs": len(set(new_blobs) - a_oids),
                   "history_only_blob_bytes": sum(v["bytes"] for k, v in new_blobs.items() if k not in a_oids)},
        "counts": dict(counts), "zip_archives": zip_details, "sqlite_databases": db_details,
        "pdf_files": pdf_details, "errors": errors, "json_parse_errors": json_errors,
        "embedded_json_depth_limits": embedded_depth_limits,
        "binary_limit": "Whole bytes hashed and ASCII credential literals scanned. No OCR, pixel interpretation, arbitrary encoded secrets, encrypted payload decryption, or unsaved traffic claim.",
    }
    refs = [dump("public-payload-A-inventory.json", inventory)]
    refs.append(dump("public-payload-A-history-inventory.json", {
        **common, "status": "FULL_CONTENT_SCAN_COMPLETE_CONTEXT_REVIEW_PENDING",
        "new_objects_counts": dict(collections.Counter(x["type"] for x in objects)),
        "new_object_metadata": objects, "new_commits": commits,
        "new_blob_count": len(new_blobs), "new_blob_bytes": sum(x["bytes"] for x in new_blobs.values()),
        "new_blobs": history,
        "normal_push_scope": "All newly reachable blobs from baseline..A, including superseded paths. Exact A blob overlap reuses this run's complete scan.",
    }))
    refs.append(dump("public-payload-A-candidates.json", {
        **common, "status": "CONTEXT_REVIEW_PENDING",
        "groups": list(candidate_groups.values()),
        "summary": dict(collections.Counter(x["classification"] for x in candidate_groups.values())),
        "raw_values_disclosed": False,
    }))
    refs.append(dump("public-payload-A-scan-execution.json", {
        **common, "command": ".venv/Scripts/python.exe artifacts/final-evidence-pack/9e83fe7/scan_public_payload_A.py",
        "counts": dict(counts), "artifacts": refs.copy(), "errors": errors,
        "global_goal": "NOT_ACHIEVED", "publication": "NOT_PERFORMED",
        "future_B_extension": "PENDING; ignored current manual/evidence/report artifacts are not implicitly included in A.",
    }))
    print(json.dumps({"artifacts": refs, "counts": dict(counts), "candidate_groups": len(candidate_groups), "json_errors": len(json_errors)}), flush=True)


if __name__ == "__main__":
    main()
