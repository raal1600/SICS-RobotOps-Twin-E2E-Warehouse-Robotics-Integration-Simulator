EVENTS = []
IDS = itertools.count(1)

ZERO = time.perf_counter()
PARENT = ContextVar("diagnostic_parent", default=None)
LOOP_ERRORS = []

@contextmanager
def measured(kind, name, **fields):
    ident = next(IDS)
    parent = PARENT.get()
    token = PARENT.set(ident)
    started = time.perf_counter()
    phase = PHASE
    error = None
    try:
        yield
    except BaseException as exc:
        error = type(exc).__name__ + ": " + str(exc)[:180]
        raise
    finally:
        ended = time.perf_counter()
        EVENTS.append({"id":ident,"parent":parent,"phase":phase,"kind":kind,"name":name,"start_ms":(started-ZERO)*1000,"end_ms":(ended-ZERO)*1000,"elapsed_ms":(ended-started)*1000,"thread":threading.current_thread().name,"error":error,**fields})
        PARENT.reset(token)

class TimedConnection(sqlite3.Connection):
    def __init__(self, database, *args, **kwargs):
        self.diagnostic_path = str(database)
        with measured("sqlite", "connect", database=self.diagnostic_path, timeout=kwargs.get("timeout", args[0] if args else 5.0)):
            super().__init__(database, *args, **kwargs)
    def execute(self, sql, parameters=(), /):
        with measured("sqlite", "execute", database=self.diagnostic_path, sql=" ".join(sql.split())[:160]):
            return super().execute(sql, parameters)
    def executescript(self, sql, /):
        with measured("sqlite", "executescript", database=self.diagnostic_path, sql=" ".join(sql.split())[:160]):
            return super().executescript(sql)
    def commit(self):
        with measured("sqlite", "commit", database=self.diagnostic_path):
            return super().commit()
    def rollback(self):
        with measured("sqlite", "rollback", database=self.diagnostic_path):
            return super().rollback()
    def close(self):
        with measured("sqlite", "close", database=self.diagnostic_path):
            return super().close()

ORIGINAL_CONNECT = sqlite3.connect
def timed_connect(*args, **kwargs):
    kwargs.setdefault("factory", TimedConnection)
    return ORIGINAL_CONNECT(*args, **kwargs)

ORIGINALS = []
def instrument(cls, name, context=False):
    original = getattr(cls, name)
    ORIGINALS.append((cls, name, original))
    if context:
        @contextmanager
        @wraps(original)
        def replacement(*args, **kwargs):
            with measured("method", cls.__name__+"."+name, exclusive=kwargs.get("exclusive")):
                with original(*args, **kwargs) as value:
                    yield value
    else:
        @wraps(original)
        def replacement(*args, **kwargs):
            with measured("method", cls.__name__+"."+name):
                return original(*args, **kwargs)
    setattr(cls, name, replacement)

class RequestTiming:
    def __init__(self, app): self.app = app
    async def __call__(self, scope, receive, send):
        if scope["type"] != "http":
            return await self.app(scope, receive, send)
        start = time.perf_counter()
        sent = []
        async def timed_send(message):
            if message["type"].startswith("http.response"):
                sent.append({"type":message["type"],"status":message.get("status"),"more_body":message.get("more_body"),"elapsed_ms":(time.perf_counter()-start)*1000})
            await send(message)
        with measured("http_server",scope["method"]+" "+scope["path"],response_messages=sent):
            await self.app(scope, receive, timed_send)

class DiagnosticServer(uvicorn.Server):
    async def serve(self, sockets=None):
        self.diagnostic_loop = asyncio.get_running_loop()
        def exception_handler(loop, context):
            LOOP_ERRORS.append({"message":context.get("message"),"exception":repr(context.get("exception"))})
            loop.default_exception_handler(context)
        self.diagnostic_loop.set_exception_handler(exception_handler)
        async def lag():
            expected = time.perf_counter()+0.01
            while True:
                await asyncio.sleep(0.01)
                now=time.perf_counter()
                LAG.append({"phase":PHASE,"ms":max(0,(now-expected)*1000)})
                expected=now+0.01
        task=asyncio.create_task(lag())
        try: await super().serve(sockets=sockets)
        finally:
            task.cancel()
            try: await task
            except asyncio.CancelledError: pass
