"""Audit a downloaded exact-source CI ZIP without rewriting original evidence."""
import argparse
import hashlib
import json
import re
import stat
import subprocess
import xml.etree.ElementTree as ET
import zipfile
from datetime import datetime, timezone
from pathlib import Path, PurePosixPath

parser = argparse.ArgumentParser()
parser.add_argument('archive', type=Path)
parser.add_argument('metadata', type=Path)
parser.add_argument('sha')
parser.add_argument('--out', type=Path, required=True)
parser.add_argument('--expected-tests', type=int, required=True)
args = parser.parse_args()
out = args.out.resolve()
out.mkdir(parents=True, exist_ok=True)
metadata = json.loads(args.metadata.read_text(encoding='utf8'))
if 'artifacts' in metadata:
    candidates = [a for a in metadata['artifacts'] if a['name'] == 'deterministic-evidence']
    assert len(candidates) == 1
    metadata = candidates[0]
assert metadata['workflow_run']['head_sha'] == args.sha
digest = hashlib.sha256(args.archive.read_bytes()).hexdigest()
assert metadata['digest'] == 'sha256:' + digest
assert metadata['size_in_bytes'] == args.archive.stat().st_size
with zipfile.ZipFile(args.archive) as archive:
    infos = archive.infolist()
    assert len(infos) < 10000
    assert sum(info.file_size for info in infos) < 600 * 1024 * 1024
    for info in infos:
        path = PurePosixPath(info.filename)
        assert not path.is_absolute() and '..' not in path.parts
        assert '\\' not in info.filename and ':' not in info.filename
        assert not stat.S_ISLNK(info.external_attr >> 16)
    manifests = []
    for info in infos:
        if re.fullmatch(r'docs/evidence/acceptance/[^/]+/manifest.json', info.filename):
            value = json.loads(archive.read(info))
            if value['source']['commit'] == args.sha:
                manifests.append((info.filename, value))
    assert len(manifests) == 1, [(name, m['source']) for name, m in manifests]
    manifest_name, manifest = manifests[0]
    assert manifest['source']['dirty'] is False
    directory = str(PurePosixPath(manifest_name).parent)
    selected = [info for info in infos if info.filename.startswith(directory + '/') and not info.is_dir()]
    saved = []
    for info in selected:
        destination = (out / info.filename).resolve()
        assert destination.is_relative_to(out)
        destination.parent.mkdir(parents=True, exist_ok=True)
        data = archive.read(info)
        destination.write_bytes(data)
        saved.append({'path': info.filename, 'sha256': hashlib.sha256(data).hexdigest(), 'bytes': len(data)})

tests = []
identities = []
for run in (1, 2):
    junit_path = out / directory / f'tests-{run}.xml'
    tree = ET.parse(junit_path)
    cases = tree.findall('.//testcase')
    assert len(cases) == args.expected_tests
    for category in ('failure', 'error', 'skipped'):
        assert not tree.findall('.//' + category)
    ids = sorted((case.get('classname'), case.get('name')) for case in cases)
    assert len(ids) == len(set(ids))
    identities.append(ids)
    coverage_path = out / directory / f'coverage-{run}.json'
    coverage = json.loads(coverage_path.read_text(encoding='utf8'))['totals']['percent_covered']
    assert coverage >= 85
    tests.append({'run': run, 'tests': len(cases), 'passed': len(cases), 'failures': 0,
                  'errors': 0, 'skipped': 0, 'coverage_percent': coverage,
                  'junit_sha256': hashlib.sha256(junit_path.read_bytes()).hexdigest(),
                  'coverage_sha256': hashlib.sha256(coverage_path.read_bytes()).hexdigest()})
assert identities[0] == identities[1]
remote_names = {'ci', 'publication_remote', 'pages'}
local_gates = {name: gate for name, gate in manifest['gates'].items() if name not in remote_names}
assert all(gate['passed'] and gate.get('exit_code', 0) == 0 for gate in local_gates.values())
assert len([name for name in local_gates if name.startswith('demo_')]) == 8
repo = Path(__file__).resolve().parents[1]
criteria_text = subprocess.check_output(['git', 'show', args.sha + ':SUCCESS_CRITERIA.md'], cwd=repo).decode('utf8')
must_ids = set(re.findall(r'\*\*((?:SC-[A-Z]+-\d{3}|HKM-VIS-MUST-\d{3}|UI-INV-MUST-\d{3})) MUST\*\*', criteria_text))
assert len(must_ids) == 118
assert must_ids <= manifest['criteria'].keys()
pending = {key: manifest['criteria'][key] for key in sorted(must_ids) if not manifest['criteria'][key]['passed']}
result = {
    'result': 'PASS_FOR_CI_LOCAL_EVIDENCE_AUDIT',
    'verified_at_utc': datetime.now(timezone.utc).isoformat(), 'source': manifest['source'],
    'artifact_id': metadata['id'], 'artifact_sha256': digest, 'artifact_digest_verified': True,
    'archive_entries_validated': len(infos), 'selected_manifest': manifest_name,
    'selected_manifest_sha256': hashlib.sha256((out / manifest_name).read_bytes()).hexdigest(),
    'environment': manifest['environment'], 'tests': tests, 'test_identities_identical': True,
    'local_gate_count': len(local_gates), 'local_gates': local_gates,
    'mandatory_criteria_count': len(must_ids), 'archived_unrefreshed_musts': pending,
    'remote_gate_placeholders': {name: manifest['gates'][name] for name in remote_names},
    'evidence_files': saved,
    'limitations': 'CI --local leaves remote gate placeholders unrefreshed. Original manifest remains byte-exact; separate workflow/Pages evidence supplies remote results. Historical manifests were excluded.'
}
(out / 'ci-evidence-verification.json').write_text(json.dumps(result, indent=2) + '\n', encoding='utf8')
print(json.dumps({key: value for key, value in result.items() if key not in ('local_gates', 'evidence_files', 'environment')}, indent=2))
